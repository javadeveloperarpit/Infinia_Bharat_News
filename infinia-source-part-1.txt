const fs = require("fs");
const path = require("path");

const ROOT = path.join(__dirname, "..");

const IGNORE = [
  "node_modules",
  ".next",
  ".git",
  "dist",
  "build",
  ".vercel",
  ".env",
  ".env.local",
  "package-lock.json",
  "infinia-source-code.txt"
];


const ALLOWED = [
  ".ts",
  ".tsx",
  ".js",
  ".jsx",
  ".json",
  ".css",
  ".md"
];


// ek part ki maximum size
const MAX_SIZE = 50000; // ~50KB


let partNumber = 1;
let output = "";

let exportedFiles = 0;


function add(text = "") {

  output += text + "\n";

}



function savePart(){

  if(output.trim().length === 0)
    return;


  const fileName =
  `infinia-source-part-${partNumber}.txt`;


  const filePath =
  path.join(ROOT,fileName);


  fs.writeFileSync(
    filePath,
    output,
    "utf8"
  );


  console.log(
    "Created:",
    fileName
  );


  partNumber++;

  output="";

}




function checkSize(){

  if(output.length >= MAX_SIZE){

    savePart();

  }

}



function createTree(dir, prefix=""){

  let tree="";


  const items = fs.readdirSync(dir);


  for(const item of items){


    if(IGNORE.includes(item))
      continue;


    const full =
    path.join(dir,item);


    const stat =
    fs.statSync(full);



    tree +=
    prefix +
    "├── " +
    item +
    "\n";


    if(stat.isDirectory()){

      tree += createTree(
        full,
        prefix+"│   "
      );

    }

  }


  return tree;

}




function scanDirectory(dir){


const items =
fs.readdirSync(dir);



for(const item of items){


if(IGNORE.includes(item))
continue;



const fullPath =
path.join(dir,item);



const stat =
fs.statSync(fullPath);



if(stat.isDirectory()){


scanDirectory(fullPath);


}

else{


exportFile(fullPath);


}


}


}




function exportFile(filePath){


const ext =
path.extname(filePath);



if(!ALLOWED.includes(ext))
return;



const size =
fs.statSync(filePath).size;



// bade files skip
if(size > 500000)
return;



let content="";


try{


content =
fs.readFileSync(
filePath,
"utf8"
);


}
catch(e){

return;

}



const relative =
path.relative(
ROOT,
filePath
);



add("\n");
add("================================================");
add("FILE PATH:");
add(relative);
add("FILE SIZE:");
add(size+" bytes");
add("================================================");
add("\n");

add(content);



exportedFiles++;


checkSize();


}




// HEADER

add("========================================");
add(" INFINIA BHARAT NEWS SOURCE EXPORT ");
add("========================================");


add("\nPROJECT STRUCTURE\n");

add(
createTree(ROOT)
);


add("\n\nSOURCE CODE\n");



// START SCAN

scanDirectory(ROOT);



// save remaining

savePart();



console.log("");
console.log(
"Total files exported:",
exportedFiles
);

"Total parts:",
partNumber-1

export async function POST(
  request: NextRequest
) {

  try {

    const body = await request.json();

    const {
      name,
      email,
      password
    } = body;



    if(
      !name ||
      !email ||
      !password
    ){

      return NextResponse.json(
        {
          success:false,
          message:"All fields are required."
        },
        {
          status:400
        }
      );

    }



    const user =
      await adminAuth.createUser({

        displayName:name,

        email,

        password

      });



    await adminDb
      .collection("users")
      .doc(user.uid)
      .set({

        uid:user.uid,

        name,

        email,

        role:"editor",

        status:"active",

        createdAt:FieldValue.serverTimestamp(),

        updatedAt:FieldValue.serverTimestamp()

      });



    return NextResponse.json({

      success:true,

      message:"Editor created successfully."

    });

  }

  catch(error:any){

    return NextResponse.json(

      {

        success:false,

        message:error.message

      },

      {

        status:500

      }

    );

  }

}

request:NextRequest
){

try{


const authHeader =

const currentUser = await verifyRole(

 ]

 uid: string;
 role: "admin" | "superAdmin";
};



const {uid}=await request.json();

if(

if(

  return (
