  thumbnail: string;

  categoryId: string;
  featured: boolean;
  breaking: boolean;
  priority: number;
  status: "draft" | "published";
  createdAt?: any;
}

export async function getPublishedArticles(): Promise<PublicArticle[]> {


const q = query(

collection(db,"articles"),

where(
"status",
"==",
"published"
),

orderBy(
"createdAt",
"desc"
),

limit(20)

);



const snap =
await getDocs(q);



return snap.docs.map(
  (doc) =>
    ({
      id: doc.id,
      ...doc.data(),
    } as PublicArticle)
